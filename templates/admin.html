const ADMIN_KEY = "afit2026"; // Ensure this matches your app.py async function updateDashboard() { try { const res = await fetch(`/admin/all_data?key=${ADMIN_KEY}`); const data = await res.json(); // Update the 6 Stat Cards document.getElementById('stat-total').textContent = data.bookings.length; document.getElementById('stat-queued').textContent = data.bookings.filter(b => b.status === 'queued').length; document.getElementById('stat-active').textContent = data.bookings.filter(b => b.status === 'assigned').length; document.getElementById('stat-free').textContent = data.tricycles.filter(t => t.status === 'free').length; // Update the Bookings Table const tableBody = document.getElementById('admin-table-body'); if (data.bookings.length === 0) { tableBody.innerHTML = 'No recent bookings'; } else { tableBody.innerHTML = data.bookings.slice(-5).reverse().map(b => ` ${b.ticket_code} ${b.name} ${b.status.toUpperCase()} `).join(''); } } catch (err) { console.error("Dashboard Sync Error:", err); } } setInterval(updateDashboard, 5000); updateDashboard();
